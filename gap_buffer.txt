#include <stdio.h>

void toString(char *buffer, unsigned int len, unsigned int *s,
unsigned int *e)
{
    *e += 1;
    while (*e < len)
    {
        printf("%d\n", *e);
        buffer[(*s)++] = buffer[(*e)++];
    }
    buffer[*s] = 0;
}

void cursorMoveLeft(char *buffer, unsigned int *s, unsigned int *e)
{
    buffer[(*e)--] = buffer[--(*s)];
    buffer[*s] = 0;
}

void insert(char c, char *buffer, unsigned int *s)
{
    buffer[(*s)++] = c;
}

void delete(char *buffer, unsigned int *s)
{
    buffer[(*s)--] = 0;
}

int main(void)
{
    char gapBuffer[32] = {0};
    unsigned int gapStart = 0;
    unsigned int gapEnd = 31;
    insert('i', gapBuffer, &gapStart);
    insert('n', gapBuffer, &gapStart);
    insert('t', gapBuffer, &gapStart);
    cursorMoveLeft(gapBuffer, &gapStart, &gapEnd);
    cursorMoveLeft(gapBuffer, &gapStart, &gapEnd);
    cursorMoveLeft(gapBuffer, &gapStart, &gapEnd);
    insert('f', gapBuffer, &gapStart);
    //toString(gapBuffer, sizeof(gapBuffer), &gapStart, &gapEnd);

    {

        {

        }

        {
            printf("_ ");
        }
    }

    return 0;
}
